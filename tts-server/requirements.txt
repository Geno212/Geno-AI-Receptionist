# Use a dedicated venv (recommended — avoids breaking other Python projects):
#   python -m venv .venv
#   .venv\Scripts\activate          # Windows
#   # source .venv/bin/activate     # macOS/Linux
#
# Install PyTorch FIRST, matched to your CUDA version, then this file:
#   pip install torch==2.6.0 torchvision==0.21.0 torchaudio==2.6.0 --index-url https://download.pytorch.org/whl/cu124
#   pip install -r requirements.txt
#   pip install --no-deps chatterbox-tts s3tokenizer
#
# torch is intentionally NOT pinned here: the correct wheel depends on the
# machine's CUDA runtime, and installing the default CPU wheel would silently
# make synthesis unusably slow.
#
# On cards with <8 GB VRAM start with: TTS_LAZY_UNLOAD=1 python server.py
